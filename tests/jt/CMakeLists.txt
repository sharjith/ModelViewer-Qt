cmake_minimum_required(VERSION 3.18)
project(JtImportTests LANGUAGES CXX)
find_package(OpenCASCADE REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(assimp REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET mv_jt_codec)
  add_subdirectory(${REPO}/third_party/occt-jt ${CMAKE_CURRENT_BINARY_DIR}/codec)
endif()
add_executable(jt_probe jt_probe.cpp ${REPO}/src/Import/Jt/JtImporter.cpp ${REPO}/src/Import/Jt/JtSceneBuilder.cpp)
target_include_directories(jt_probe PRIVATE ${REPO}/include/Import)
target_link_libraries(jt_probe PRIVATE mv_jt_codec Qt6::Core assimp::assimp ${OpenCASCADE_ModelingData_LIBRARIES} ${OpenCASCADE_ModelingAlgorithms_LIBRARIES})
add_executable(jt_tests jt_tests.cpp jt_scene_tests.cpp ${REPO}/src/Import/Jt/JtImporter.cpp ${REPO}/src/Import/Jt/JtSceneBuilder.cpp)
target_include_directories(jt_tests PRIVATE ${REPO}/include/Import ${REPO}/src/Import/Jt)
target_link_libraries(jt_tests PRIVATE mv_jt_codec Qt6::Core assimp::assimp ${OpenCASCADE_ModelingData_LIBRARIES} ${OpenCASCADE_ModelingAlgorithms_LIBRARIES})
enable_testing()
add_test(NAME jt_codec_tests COMMAND jt_tests)
option(MV_JT_SANITIZE "Instrument the isolated JT decoder" OFF)
if(MV_JT_SANITIZE AND MSVC)
  target_compile_options(mv_jt_codec PRIVATE /fsanitize=address /Zi)
  target_compile_options(jt_probe PRIVATE /fsanitize=address /Zi)
  target_compile_options(jt_tests PRIVATE /fsanitize=address /Zi)
  target_link_options(jt_probe PRIVATE /DEBUG /INCREMENTAL:NO)
  target_link_options(jt_tests PRIVATE /DEBUG /INCREMENTAL:NO)
endif()
