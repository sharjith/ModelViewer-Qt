cmake_minimum_required(VERSION 3.18)
project(JtImportTests LANGUAGES CXX)
find_package(OpenCASCADE REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
find_package(assimp REQUIRED)
find_package(glm CONFIG REQUIRED)
add_executable(jt_fixturegen jt_fixturegen.cpp)
target_link_libraries(jt_fixturegen PRIVATE Qt6::Core)
target_compile_features(jt_fixturegen PRIVATE cxx_std_17)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET mv_jt_codec)
  add_subdirectory(${REPO}/src/ThirdParty/occt-jt ${CMAKE_CURRENT_BINARY_DIR}/codec)
endif()
add_executable(jt_probe jt_probe.cpp ${REPO}/src/Import/Jt/JtImporter.cpp ${REPO}/src/Import/Jt/JtSceneBuilder.cpp)
target_include_directories(jt_probe PRIVATE ${REPO}/include/Import)
target_link_libraries(jt_probe PRIVATE mv_jt_codec Qt6::Core Qt6::Gui assimp::assimp ${OpenCASCADE_ModelingData_LIBRARIES} ${OpenCASCADE_ModelingAlgorithms_LIBRARIES})
add_executable(jt_color_fixturegen jt_color_fixturegen.cpp ${REPO}/src/Import/Jt/JtImporter.cpp ${REPO}/src/Import/Jt/JtSceneBuilder.cpp)
target_include_directories(jt_color_fixturegen PRIVATE ${REPO}/include/Import)
target_link_libraries(jt_color_fixturegen PRIVATE mv_jt_codec Qt6::Core Qt6::Gui assimp::assimp ${OpenCASCADE_ModelingData_LIBRARIES} ${OpenCASCADE_ModelingAlgorithms_LIBRARIES})
add_executable(jt9_colour_fixturegen jt9_colour_fixturegen.cpp ${REPO}/src/Import/Jt/JtImporter.cpp ${REPO}/src/Import/Jt/JtSceneBuilder.cpp)
target_include_directories(jt9_colour_fixturegen PRIVATE ${REPO}/include/Import)
target_link_libraries(jt9_colour_fixturegen PRIVATE mv_jt_codec Qt6::Core Qt6::Gui assimp::assimp ${OpenCASCADE_ModelingData_LIBRARIES} ${OpenCASCADE_ModelingAlgorithms_LIBRARIES})
add_executable(jt_uv_fixturegen jt_uv_fixturegen.cpp ${REPO}/src/Import/Jt/JtImporter.cpp ${REPO}/src/Import/Jt/JtSceneBuilder.cpp ${REPO}/src/Scene/SceneUtils.cpp)
target_include_directories(jt_uv_fixturegen PRIVATE ${REPO}/include/Import ${REPO}/include/Scene)
target_link_libraries(jt_uv_fixturegen PRIVATE mv_jt_codec Qt6::Core Qt6::Gui assimp::assimp glm::glm ${OpenCASCADE_ModelingData_LIBRARIES} ${OpenCASCADE_ModelingAlgorithms_LIBRARIES})
add_executable(jt_coordinate_fixturegen jt_coordinate_fixturegen.cpp ${REPO}/src/Import/Jt/JtImporter.cpp ${REPO}/src/Import/Jt/JtSceneBuilder.cpp ${REPO}/src/Scene/SceneUtils.cpp)
target_include_directories(jt_coordinate_fixturegen PRIVATE ${REPO}/include/Import ${REPO}/include/Scene)
target_link_libraries(jt_coordinate_fixturegen PRIVATE mv_jt_codec Qt6::Core Qt6::Gui assimp::assimp glm::glm ${OpenCASCADE_ModelingData_LIBRARIES} ${OpenCASCADE_ModelingAlgorithms_LIBRARIES})
add_executable(jt_tests jt_tests.cpp jt_arithmetic_count_tests.cpp jt_coordinate_tests.cpp jt_uv_tests.cpp jt_huffman_tests.cpp jt_quantized_colour_tests.cpp jt9_colour_tests.cpp jt95_tests.cpp jt_scene_tests.cpp jt_legacy_tests.cpp jt_quantized_tests.cpp jt_modern_attribute_tests.cpp jt_palette_tests.cpp ${REPO}/src/Import/Jt/JtImporter.cpp ${REPO}/src/Import/Jt/JtSceneBuilder.cpp)
target_sources(jt_tests PRIVATE jt_texture_tests.cpp)
target_sources(jt_tests PRIVATE jt_texture_transform_tests.cpp)
target_sources(jt_tests PRIVATE jt_texture_planar_tests.cpp)
target_sources(jt_tests PRIVATE jt_texture_pixel_tests.cpp)
target_sources(jt_tests PRIVATE jt9_texture_tests.cpp)
target_sources(jt_tests PRIVATE jt_texture_border_tests.cpp)
target_sources(jt_tests PRIVATE ${REPO}/src/Scene/SceneUtils.cpp)
target_include_directories(jt_tests PRIVATE ${REPO}/include/Import ${REPO}/src/Import/Jt ${REPO}/include/Scene)
target_link_libraries(jt_tests PRIVATE mv_jt_codec Qt6::Core Qt6::Gui assimp::assimp glm::glm ${OpenCASCADE_ModelingData_LIBRARIES} ${OpenCASCADE_ModelingAlgorithms_LIBRARIES})
enable_testing()
add_test(NAME jt_codec_tests COMMAND jt_tests)
option(MV_JT_SANITIZE "Instrument the isolated JT decoder" OFF)
if(MV_JT_SANITIZE AND MSVC)
  target_compile_options(mv_jt_codec PRIVATE /fsanitize=address /Zi)
  target_compile_options(jt_uv_fixturegen PRIVATE /fsanitize=address /Zi)
  target_compile_options(jt_coordinate_fixturegen PRIVATE /fsanitize=address /Zi)
  target_link_options(jt_uv_fixturegen PRIVATE /DEBUG /INCREMENTAL:NO)
  target_link_options(jt_coordinate_fixturegen PRIVATE /DEBUG /INCREMENTAL:NO)
  target_compile_options(jt_probe PRIVATE /fsanitize=address /Zi)
  target_compile_options(jt_tests PRIVATE /fsanitize=address /Zi)
  target_compile_options(jt9_colour_fixturegen PRIVATE /fsanitize=address /Zi)
  target_compile_options(jt_color_fixturegen PRIVATE /fsanitize=address /Zi)
  target_link_options(jt_probe PRIVATE /DEBUG /INCREMENTAL:NO)
  target_link_options(jt_tests PRIVATE /DEBUG /INCREMENTAL:NO)
  target_link_options(jt9_colour_fixturegen PRIVATE /DEBUG /INCREMENTAL:NO)
  target_link_options(jt_color_fixturegen PRIVATE /DEBUG /INCREMENTAL:NO)
endif()
